            SOLID Liskov Substitution Principle (LSP):
            

              The Dialogue Manager downstream receives a uniform NLPResult(intent, entities, tool_call) regardless of whether it came from a 50-line regex module or a 70-billion-parameter LLM.
            

          

        

      


      
      

        

          
          INLPService Interface & Data Transfer Object
        

        
class INLPService(ABC):
    @abstractmethod
    def parse(self, text: str) -> NLPResult:
        """Parses raw text into intent, entities, sentiment, and optional tool call."""
        pass

class NLPResult(BaseModel):
    raw_text: str
    intent: str
    confidence: float = 1.0
    entities: Dict[str, Any] = Field(default_factory=dict)
    sentiment: float = 0.0  # -1.0 to 1.0
    tool_call: Optional[Dict[str, Any]] = None
    provider: str = "rule_based"
